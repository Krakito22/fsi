import tkinter as tk
from tkinter import PhotoImage

WIDTH, HEIGHT = 700, 500
MOB_SPEED = 5  
DIRECTION = 1 

class Mob:
    def __init__(self, canvas, x, y, image):
        self.canvas = canvas
        self.x, self.y = x, y
        self.life = 1
        self.image = canvas.create_image(x, y, image=image)

    def move(self):
        global DIRECTION
        self.canvas.move(self.image, MOB_SPEED * DIRECTION, 0)
        self.x += MOB_SPEED * DIRECTION

    def hit(self):
        self.life -= 1
        if self.life <= 0:
            self.canvas.delete(self.image)
            mobs.remove(self)
            check_victory()


class Player:
    def __init__(self, canvas, x, y, image):
        self.canvas = canvas
        self.image = canvas.create_image(x, y, image=image)
        self.x = x

    def move(self, dx):
        self.x += dx
        self.canvas.move(self.image, dx, 0)


class Bullet:
    def __init__(self, canvas, x, y, image):
        self.canvas = canvas
        self.x, self.y = x, y
        self.image = canvas.create_image(x, y, image=image)
        self.move_bullet()

    def move_bullet(self):
        """Déplace la balle vers le haut et vérifie les collisions."""
        self.canvas.move(self.image, 0, -10)
        self.y -= 10
        
       
        for mob in mobs:
            if abs(self.x - mob.x) < 25 and abs(self.y - mob.y) < 25:  
                mob.hit()
                self.destroy()
                return

        
        if self.y < 0:
            self.destroy()
        else:
            self.canvas.after(50, self.move_bullet)

    def destroy(self):
        """Supprime la balle du canvas et de la liste des balles."""
        self.canvas.delete(self.image)
        bullets.remove(self)


def check_victory():
    if not mobs:
        canvas.create_text(WIDTH//2, HEIGHT//2, text="VICTOIRE !", fill="white", font=("Arial", 24))


def move_mobs():
    global DIRECTION
    for mob in mobs:
        mob.move()

   
    leftmost = min(mob.x for mob in mobs) if mobs else WIDTH
    rightmost = max(mob.x for mob in mobs) if mobs else 0
    if leftmost <= 50 or rightmost >= WIDTH - 50:
        DIRECTION *= -1  

    canvas.after(500, move_mobs)  

def keypress(event):
    if event.keysym == "Right":
        player.move(20)
    elif event.keysym == "Left":
        player.move(-20)
    elif event.keysym == "space":
        bullets.append(Bullet(canvas, player.x, 400, bullet_image))
    elif event.keysym == "Escape":
        pause_menu()

def pause_menu():
    pause = tk.Toplevel(main)
    pause.geometry("150x100+275+200")
    pause.title("Pause")
    pause.configure(background="black")
    pause.overrideredirect(True)
    tk.Button(pause, text="Reprendre", command=pause.destroy, bg="white", fg="black").pack()
    tk.Button(pause, text="Quitter", command=main.quit, bg="red", fg="white").pack()
    pause.grab_set()

main = tk.Tk()
main.title("Space Invaders")
main.geometry(f"{WIDTH}x{HEIGHT}")
main.resizable(False, False)
main.configure(bg="black")

canvas = tk.Canvas(main, width=WIDTH, height=HEIGHT)
canvas.pack()

space_bg = PhotoImage(file=r"C:/Users/mikak/Desktop/Python/Projet Python SI/space_background.png")
mob_image = PhotoImage(file=r"C:/Users/mikak/Desktop/Python/Projet Python SI/mob.png")
player_image = PhotoImage(file=r"C:/Users/mikak/Desktop/Python/Projet Python SI/player.png")
bullet_image = PhotoImage(file=r"C:/Users/mikak/Desktop/Python/Projet Python SI/bullet.png")

canvas.create_image(WIDTH//2, HEIGHT//2, image=space_bg)

player = Player(canvas, WIDTH // 2, 450, player_image)

mobs = [Mob(canvas, x, 100, mob_image) for x in range(150, 650, 100)]

bullets = []

main.bind("<Key>", keypress)

move_mobs()

main.mainloop()
